# FGUI 多语言插件实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** FGUI 声明式多语言：编辑器插件把 key 写入组件 customData 的 `L10n:` 段，运行时包构造时解析并自动应用文本，语言切换经 WinBase 遍历重应用。

**Architecture:** FairyGUI（AOT）以 partial 分部（`Scripts/UI/CustomExt/`）承载全部 L10n 逻辑，原版文件仅 3 个单行锚点；文本解析经 `GObject.GetLanguageText` 静态委托（`Func<string,string>`）由热更层注入（AOT/热更两阶段）；刷新挂在 `WinBase._OnLanguageChanged`（可见界面 `OnOpen()` + `RefreshAllL10n(WinUI)` 双轨）。

**Tech Stack:** FairyGUI（原版结构源码，`Scripts/UI/`）、partial class、字符串 key（`L10nKey`/`TbLocalization`）。

**Spec:** `Docs/superpowers/specs/2026-09-19-fgui-l10n-design.md`

**工作流（用户指令）:** 所有改动只写文件不提交；全部完成并验证完毕后询问用户，经确认后统一整理提交。

## Global Constraints

- 段前缀统一 `L10n:`（编辑器写入 = 运行时解析，不用参考项目的 `L10n&`）；customData 按 `|` 分段共存
- key 语义为**字符串**；key 查不到时沿用 `LocalizationModule.GetLanguageText` 现行为（LogWarning + `[key]`）
- 委托签名固定 `Func<string, string>`；两阶段注入：AOT 注 `LaunchLocalization.GetLanguage`，热更注 `LocalizationModule.GetLanguageText`（覆盖式）
- Unity 运行时铁律：禁 Task/Coroutine/LINQ/运行时反射；FairyGUI 源码（`Unity/Assets/3rdPlugins/FairyGUI/`）为 AOT 程序集，同样遵守
- 分部文件放 `Scripts/UI/CustomExt/`，namespace 保持 `FairyGUI`；原版文件改动仅限计划列出的锚点行
- 编辑器插件（`Tools/FairyGUI-Editor-master/plugin/CustomL10n/`，Lua）由用户主导开发；本计划覆盖运行时链与联调
- Unity 编译验证：`unity-cli tool call refresh_assets` → 等 14s → `unity-cli tool call get_compilation_state`，期望 `errorCount: 0`

**计划关键事实（已核实）**: 原版 `GObject.Setup_BeforeAdd` 末尾（`GObject.cs:1969-1971`）已把编辑器 customData（字符串索引 `ReadS()`）读入 **`public object data` 字段**——所有组件类型通用，无需动任何包读取点；`GComponent.cs:1535` 的 `Skip(2)` 注释有误导，与编辑器 customData 无关，保持不动。

- [ ] **Step 1: 写 `GObject.L10n.cs`**

```csharp
using System;
using System.Collections.Generic;

namespace FairyGUI
{
    /// <summary>
    /// L10n 段解析结果。
    ///     简单模式：IsCtrl=false，SimpleKey 为多语言 key；
    ///     控制器模式：IsCtrl=true，CtrlName 为父组件控制器名，PageKeys 为 页索引 → key。
    /// </summary>
    internal class L10nGearData
    {
        public bool IsCtrl;
        public string SimpleKey;
        public string CtrlName;
        public Dictionary<int, string> PageKeys;
    }

    /// <summary>
    /// GObject 的 L10n（声明式多语言）扩展。
    ///     数据来源：FGUI 编辑器 CustomL10n 插件写入组件 customData 的 "L10n:" 段
    ///     （customData 已由原版 Setup_BeforeAdd 读入 this.data 字段）。
    ///     应用时机：Setup_AfterAdd 末尾（控制器/子级就绪后）+ 控制器切页 + RefreshAllL10n 遍历。
    /// </summary>
    public partial class GObject
    {
        /// <summary>
        /// 语言文本解析委托：由热更层启动时注入（AOT 阶段注 AOT 表版本，热更后覆盖为热更表版本）。
        /// 未注入时 ResolveL10nText 原样返回 key。
        /// </summary>
        public static Func<string, string> GetLanguageText;

        /// <summary>
        /// L10n 段解析结果（SetupL10nGear 时填充）。
        /// </summary>
        internal L10nGearData _l10nGear;

        /// <summary>
        /// 控制器模式的页变更回调引用（DisposeL10n 时退订，防泄漏）。
        /// </summary>
        private GController _l10nController;
        private EventCallback0 _l10nCtrlCallback;

        /// <summary>
        /// 是否处理 L10n：仅 GTextField/GButton/GLabel 覆写为 true，其余组件零开销跳过。
        /// </summary>
        virtual protected bool ShouldHandleL10nGear() => false;

        /// <summary>
        /// 把解析出的 key 对应文本写入目标属性（子类覆写：text/title）。
        /// </summary>
        /// <param name="key">多语言 key（已解析为当前页/简单模式）</param>
        virtual protected void OnApplyL10nText(string key)
        {
        }

        /// <summary>
        /// 从 customData 字符串解析 L10n: 段（customData 按 | 分段，L10n: 段与其他插件段共存）。
        /// </summary>
        /// <param name="customData">组件 customData 字符串</param>
        /// <returns>解析结果；无 L10n 段时返回 null</returns>
        internal static L10nGearData ParseL10nGear(string customData)
        {
            if (string.IsNullOrEmpty(customData)) return null;

            foreach (var segment in customData.Split('|'))
            {
                if (!segment.StartsWith("L10n:")) continue;

                var body = segment.Substring(5);
                if (body.IndexOf(',') >= 0 && body.IndexOf('=') >= 0)
                {
                    // 控制器模式：{ctrlName},{pageId}={key},{pageId}={key}...
                    var data = new L10nGearData { IsCtrl = true, PageKeys = new Dictionary<int, string>() };
                    var sepIndex = body.IndexOf(',');
                    data.CtrlName = body.Substring(0, sepIndex);
                    foreach (var pair in body.Substring(sepIndex + 1).Split(','))
                    {
                        var kv = pair.Split('=');
                        if (kv.Length == 2 && int.TryParse(kv[0], out int pageId))
                            data.PageKeys[pageId] = kv[1];
                    }
                    return data.PageKeys.Count > 0 ? data : null;
                }

                return string.IsNullOrEmpty(body) ? null : new L10nGearData { SimpleKey = body };
            }

            return null;
        }

        /// <summary>
        /// Setup_AfterAdd 锚点入口：解析 data 中的 L10n 段并首次应用，控制器模式订阅页变更。
        /// </summary>
        internal void SetupL10nGear()
        {
            DisposeL10n(); // 池化复用/重复 Setup 防御：先重置

            if (!ShouldHandleL10nGear()) return;

            _l10nGear = ParseL10nGear(this.data as string);
            if (_l10nGear == null) return;

            ApplyL10nGearData();

            if (_l10nGear.IsCtrl && parent is GComponent parentComp)
            {
                var ctrl = parentComp.GetController(_l10nGear.CtrlName);
                if (ctrl != null)
                {
                    _l10nController = ctrl;
                    _l10nCtrlCallback = ApplyL10nGearData;
                    ctrl.onStateChanged.Add(_l10nCtrlCallback);
                }
            }
        }

        /// <summary>
        /// GComponent 锚点入口：对全部子组件递归补调 SetupL10nGear（控制器/子级就绪后统一触发）。
        /// </summary>
        internal void ApplyChildrenL10n()
        {
            int cnt = numChildren;
            for (int i = 0; i < cnt; i++)
            {
                var child = GetChildAt(i);
                child.SetupL10nGear();
                if (child is GComponent childComp)
                    childComp.ApplyChildrenL10n();
            }
        }

        /// <summary>
        /// 按解析结果应用文本：简单模式直取 key；控制器模式按当前页取 key。
        /// </summary>
        internal void ApplyL10nGearData()
        {
            if (_l10nGear == null) return;

            string key;
            if (_l10nGear.IsCtrl)
            {
                var pageIndex = (parent is GComponent parentComp
                    ? parentComp.GetController(_l10nGear.CtrlName)?.selectedIndex : null) ?? -1;
                if (!_l10nGear.PageKeys.TryGetValue(pageIndex, out key)) return;
            }
            else
            {
                key = _l10nGear.SimpleKey;
            }

            if (string.IsNullOrEmpty(key)) return;
            OnApplyL10nText(ResolveL10nText(key));
        }

        /// <summary>
        /// key → 文本：委托未注入时原样返回 key。
        /// </summary>
        /// <param name="key">多语言 key</param>
        /// <returns>对应语言的文本</returns>
        internal string ResolveL10nText(string key)
        {
            var getter = GetLanguageText;
            return getter == null ? key : (getter(key) ?? key);
        }

        /// <summary>
        /// 语言切换遍历入口：对 root 子树内已绑定 L10n 的组件重应用文本。
        /// 由 Hotfix 侧 WinBase 在语言切换事件中调用（仅可见界面）。
        /// </summary>
        /// <param name="root">界面根组件</param>
        public static void RefreshAllL10n(GComponent root)
        {
            if (root == null) return;

            if (root._l10nGear != null)
                root.ApplyL10nGearData();

            int cnt = root.numChildren;
            for (int i = 0; i < cnt; i++)
            {
                if (root.GetChildAt(i) is GComponent childComp)
                    RefreshAllL10n(childComp);
                else if (root.GetChildAt(i)._l10nGear != null)
                    root.GetChildAt(i).ApplyL10nGearData();
            }
        }

        /// <summary>
        /// 释放 L10n 状态并退订控制器页变更（GObject.Dispose 锚点调用）。
        /// </summary>
        internal void DisposeL10n()
        {
            if (_l10nController != null && _l10nCtrlCallback != null)
                _l10nController.onStateChanged.Remove(_l10nCtrlCallback);
            _l10nController = null;
            _l10nCtrlCallback = null;
            _l10nGear = null;
        }
    }
}
```
